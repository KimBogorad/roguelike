package roguelike;

import java.io.File;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Scanner;

import roguelike.model.units.*;
import roguelike.world.*;
import roguelike.ui.GameView;

public class GameManager {

    private static final int MAX_LEVELS = 4;
    private static final String LEVEL_FILE_PREFIX = "level";
    private static final String LEVEL_FILE_SUFFIX = ".txt";

    private GameBoard board;
    private Hero player;
    private List<Enemy> enemies;
    
    
    private int currentLevel;
    private String levelsDir;
    private final EnemyFactory enemyFactory;
    private final HeroFactory heroFactory;
    private final GameView view;

    /** Inject a custom output sink. */
    public GameManager(String path, GameView view) {
        this.enemies = new ArrayList<>();
        this.currentLevel = 1;
        this.levelsDir = path;
        this.enemyFactory = new EnemyFactory();
        this.heroFactory = new HeroFactory();
        this.view = view;
    }

    // -------------------------------------------------------------------------
    // Public entry point
    // -------------------------------------------------------------------------

    public void startGame() {
        boolean playAgain = true;
        while (playAgain) {
            this.currentLevel = 1;
            this.enemies.clear();
            view.clearScreen();
            view.displayMessage("Welcome to D&D!");
            selectPlayer();
            while (currentLevel <= MAX_LEVELS && isPlayerAlive()) {
                String levelFile;
                if (levelsDir.isEmpty()) {
                    levelFile = LEVEL_FILE_PREFIX + currentLevel + LEVEL_FILE_SUFFIX;
                } else {
                    levelFile = levelsDir + File.separator + LEVEL_FILE_PREFIX + currentLevel + LEVEL_FILE_SUFFIX;
                }
                if (!loadLevel(levelFile)) {
                    view.displayMessage("Error: Could not find or load -> " + levelFile);
                    view.displayMessage("Please check if the file path is correct. Exiting game.");
                    return;
                }
                view.displayMessage("--- Loading Level " + currentLevel + " ---");
                playLevel();

                if (isPlayerAlive()) {
                    currentLevel++;
                }
            }
            view.displayMessage(isPlayerAlive() ? "You won!" : "Game Over.");
            boolean validChoice = false;
            while (!validChoice) {
                view.displayMessage("Do you want to restart (y/n)?");
                view.displayMessage("If not there will be consequences Muwhahaha.");
                char choice = ' ';
                try {
                    choice = view.readChar();
                } catch (IllegalArgumentException e) {
                    view.displayMessage("Invalid input. Please enter a single character.");
                    continue;
                }
                switch (choice) {
                    case 'y' -> {
                        view.clearScreen();
                        validChoice = true;
                    }
                    case 'n' -> {
                        view.displayMessage("Thank you for playing, deleting system32...");
                        validChoice = true;
                        playAgain = false;
                    }
                    default -> {
                        view.displayMessage("Invalid input! Please enter exactly 'y' or 'n'.");
                    }
                }
            }
        }
    }

    // -------------------------------------------------------------------------
    // Player selection
    // -------------------------------------------------------------------------

    private void selectPlayer() {
        view.displayMessage("Select player:");
        view.displayMessage("1. Jon Snow     (Warrior)");
        view.displayMessage("2. The Hound    (Warrior)");
        view.displayMessage("3. Melisandre   (Mage)");
        view.displayMessage("4. Thoros of Myr (Mage)");
        view.displayMessage("5. Arya Stark   (Rogue)");
        view.displayMessage("6. Bronn        (Rogue)");
        view.displayMessage("7. Ygritte      (Hunter)");

        int choice = -1;
        boolean validChoice = false;
        while (!validChoice){
            try {
                choice = view.readInt();
                if (choice >= 1 && choice <= 7){
                    validChoice = true;
                } else {
                    view.displayMessage("Invalid choice. Please enter a number between 1 and 7.");
                }

            } catch (NumberFormatException e) {
                view.displayMessage("Invalid input. Please enter a valid number.");
            }
        }
        player = heroFactory.createPlayer(choice);
        player.setMessageCallback(m -> view.displayMessage(m));
    }

    

    // -------------------------------------------------------------------------
    // Game loop
    // -------------------------------------------------------------------------

    private void playLevel() {
        while (isPlayerAlive() && !enemies.isEmpty()) {
            view.drawBoard(board.toString(), player.description());
            char input = ' ';
            try {
                input = view.readChar();
            } catch (IllegalArgumentException e) {
                view.displayMessage("Invalid input. Please enter a single character.");
                continue;
            } 
            view.clearScreen();
            boolean validTurn = processPlayerInput(input);
            if (validTurn) {
                player.Tick();
                tickEnemies();
                enemies.removeIf(e -> {
                    if (!e.isAlive()) {
                        Position p = board.getPositionOf(e);
                        if (p != null) {
                            board.setOccupant(p, new Occupant());
                        }
                        return true;
                    }
                    return false;
                });
            }
        }
    }

    /** Tick every living enemy: move monsters, trigger traps, double dispatch. */
    private void tickEnemies() {
        for (Enemy enemy : enemies) {
            if (enemy.isAlive()) {
                enemy.processTurn(board, player);
            }
        }
    }

    // -------------------------------------------------------------------------
    // Input handling
    // -------------------------------------------------------------------------

    private boolean processPlayerInput(char input) {
        return switch (input) {
            case 'w' -> movePlayer(0, -1);
            case 's' -> movePlayer(0,  1);
            case 'a' -> movePlayer(-1, 0);
            case 'd' -> movePlayer(1,  0);
            case 'e' -> castAbility();
            case 'q' -> true;
            default  -> { 
                view.displayMessage("Invalid input. Please use w/a/s/d/e/q.");
                yield false;
            }
        };
    }

    private boolean movePlayer(int dx, int dy) {
        Position current = board.getPositionOf(player);
        Position target  = new Position(current.GetX() + dx, current.GetY() + dy);
        final boolean[] validMove = {true};
        board.getCell(target).accept(new CellVisitor() {
            @Override
            public void visit(Wall wall) {
                view.displayMessage("You cannot move there, it's a wall!");
                validMove[0] = false;
            }
            @Override
            public void visit(Floor floor) {
                board.attemptMove(player, target);
            }
        });
        return validMove[0];
    }

    // Helper Function: use stream() to create an arraylist of all enemies in hero's ability_range.
    // Sorted by proximity (closer range)
    // return true if ability was cast, false otherwise
    private boolean castAbility() {
        Position playerPos = board.getPositionOf(player);

        List<Enemy> enemiesInRange = enemies.stream()
                .filter(e -> playerPos.Range(board.getPositionOf(e)) < player.getAbilityRange())
                .sorted(Comparator.comparingDouble(e -> playerPos.Range(board.getPositionOf(e))))
                .collect(java.util.stream.Collectors.toCollection(ArrayList::new));

        return player.CastAbility(enemiesInRange, player);
    }

    // -------------------------------------------------------------------------
    // Level loading
    // -------------------------------------------------------------------------

    private boolean loadLevel(String filePath) {
        List<String> lines = readLevelFile(filePath);
        if (lines == null || lines.isEmpty()) return false;

        int rows = lines.size();
        int cols = lines.get(0).length();
        Cell[][] cells = new Cell[rows][cols];
        enemies.clear();

        for (int y = 0; y < rows; y++) {
            String line = lines.get(y);
            for (int x = 0; x < cols; x++) {
                cells[y][x] = parseCell(line.charAt(x), new Position(x, y));
            }
        }

        this.board = new GameBoard(cells);
        return true;
    }

    /** Returns the file's lines, or null on failure. */
    private List<String> readLevelFile(String filePath) {
        File file = new File(filePath);
        if (!file.exists()) return null;

        List<String> lines = new ArrayList<>();
        try (Scanner fileScanner = new Scanner(file)) {
            while (fileScanner.hasNextLine()) {
                String line = fileScanner.nextLine();
                if (!line.trim().isEmpty()){
                    lines.add(line);
                }
            }
        } catch (Exception e) {
            view.displayMessage("Error reading level file: " + e.getMessage());
            return null;
        }
        return lines;
    }

    /** Convert a single map character into the correct Cell. */
    private Cell parseCell(char c, Position pos) {
        if (c == '#') return new Wall(pos);
        Floor floor = new Floor(pos);
        if (c == '@') {
            placeUnit(floor, player);
        } else if (c != '.') {
            Enemy enemy = enemyFactory.createEnemy(c);
            if (enemy != null) {
                enemy.setMessageCallback(m -> view.displayMessage(m));
                enemies.add(enemy);
                placeUnit(floor, enemy);
            }
        }
        return floor;
    }

    /** Helper: wrap a unit in an Occupant and set it on the floor tile. */
    private void placeUnit(Floor floor, Unit unit) {
        Occupant occ = new Occupant();
        occ.SetUnit(unit);
        floor.SetOccupant(occ);
    }

    // -------------------------------------------------------------------------
    // Utility
    // -------------------------------------------------------------------------

    private boolean isPlayerAlive() {
        return player.isAlive();
    }
}

public class Boss extends Monster implements HeroicUnit {
    private int ability_frequency;
    private int combat_ticks;


    public Boss(char tile, String name, int hp_pool, int atk_pts, int def_pts, int exp_val, int vision_range, int ability_frequency) {
        super(tile, name, hp_pool, atk_pts, def_pts, exp_val, vision_range);
        this.ability_frequency = ability_frequency;
        this.combat_ticks = 0;
    }

    @Override
    public boolean CastAbility(List<Enemy> validTargets, Hero player) {
        sendMsg(this.name + " shoots a devastating blast at " + player.name + "!");
        int abilityDamage = this.atk_pts;
        int defenseRoll = this.rand.nextInt(player.def_pts + 1);
        return Attack(player, abilityDamage, defenseRoll);
    }

   @Override
    public void processTurn(GameBoard board, Hero player) {
        Position currentPos = board.getPositionOf(this);
        Position playerPos = board.getPositionOf(player);
        if (currentPos.Range(playerPos) < GetVisionRange()) {
            if (combat_ticks == ability_frequency) {
                combat_ticks = 0;
                CastAbility(null, player);
                return;
            } else {
                combat_ticks++;
            }
        } else {
            combat_ticks = 0;
        }
        Position targetPos = super.Move(currentPos, playerPos);
        board.attemptMove(this, targetPos);
    }
}

public abstract class Hero extends Unit implements HeroicUnit  {
    protected int exp;
    protected int player_lvl;
    protected int ability_range;
    
    public Hero(String name, int hp_pool, int atk_pts, int def_pts) {
        super(name, hp_pool, atk_pts, def_pts);
        this.exp = 0;
        this.player_lvl = 1;
    } 

    @Override
    public void Tick() {
        // request input and move accordingly. If the hero is in range of an enemy, attack it.
    }

    public void GainExp(int exp) {
        this.exp += exp;
        while (this.exp >= 50*player_lvl) {
            this.exp -= 50*player_lvl;
            player_lvl++;
            this.hp_pool += 10*player_lvl;
            this.hp_current = this.hp_pool;
            this.atk_pts += 4*player_lvl;
            this.def_pts += player_lvl;
        }
    }

    public int getAbilityRange() {
        return ability_range;
    }

    @Override
    public String description() {
        return String.format("Name: %s | HP: %d/%d | ATK: %d | DEF: %d | Level: %d | EXP: %d",
                name, hp_current, hp_pool, atk_pts, def_pts, player_lvl, exp);
    }
    
    @Override
    public String toString() {
        if(this.hp_current > 0) {
            return "@";
        }
        return "X";
    }
    
    @Override
    public void accept(OccupantVisitor visitor) {
        visitor.visit(this);
    }
}

public class Hunter extends Hero{
    private int arrows;
    private int tick_count;
    private final int STARTING_ARROWS = 10;

    public Hunter(String name, int hp_pool, int atk_pts, int def_pts, int range) {
        super(name, hp_pool, atk_pts, def_pts);
        this.ability_range = range;
        arrows = STARTING_ARROWS;
        tick_count = 0;
    }

    @Override
    public void GainExp(int exp) {
        int current_lvl = this.player_lvl;
        super.GainExp(exp);
        while(current_lvl < this.player_lvl) {
            current_lvl++;
            arrows += 10*current_lvl;
            this.atk_pts += 2*current_lvl;
            def_pts += current_lvl;
        }
    }

    @Override
    public void Tick() {
        tick_count++;
        if (tick_count == 10) {
            tick_count = 0;
            arrows += this.player_lvl;
        }
    }

    @Override
    public String description() {
        return super.description() + String.format(" | Arrows: %d", arrows);
    }

    @Override
    public boolean CastAbility(List<Enemy> validTargets, Hero player) {
        if (arrows > 0) {
            arrows--;            
            if (!validTargets.isEmpty()) {
                Enemy target = validTargets.get(0);
                int defenseRoll = this.rand.nextInt(target.def_pts + 1);
                boolean killed = Attack(target, this.atk_pts, defenseRoll);
                if (killed) {
                    this.GainExp(target.GetExpVal());
                }
            }
        }
        else {
            sendMsg(this.name + " tried to shoot, but has no arrows left!");
            return false;
        }
        return true; 
    }
}

public class Mage extends Hero {
    private int mana_pool;
    private int current_mana;
    private int mana_cost;
    private int spell_power;
    private int hits_count;

    public Mage(String name, int hp_pool, int atk_pts, int def_pts, int mana_pool, int mana_cost, int spell_power, int hits_count, int ability_range) {
        super(name, hp_pool, atk_pts, def_pts);
        this.mana_pool = mana_pool;
        current_mana = mana_pool/4;
        this.mana_cost = mana_cost;
        this.spell_power = spell_power;
        this.hits_count = hits_count;
        this.ability_range = ability_range;
    }

    @Override
    public void GainExp(int exp) {
        int current_lvl = this.player_lvl;
        super.GainExp(exp);
        while(current_lvl < this.player_lvl) {
            current_lvl++;
            mana_pool += 25*current_lvl;
            current_mana = Math.min(current_mana + mana_pool/4, mana_pool);
            spell_power += 10*current_lvl;
        }
    }

    @Override
    public void Tick() {
        super.Tick();
        current_mana = Math.min(current_mana + this.player_lvl, mana_pool);
    }

    public String description() {
        return super.description() + String.format(" | Mana: %d/%d | Spell Power: %d", current_mana, mana_pool, spell_power);
    }

    @Override
    public boolean CastAbility(List<Enemy> validTargets, Hero player) {
        if (current_mana >= mana_cost) {
            current_mana -= mana_cost;
            sendMsg(this.name + " cast Blizzard, costing " + mana_cost + " mana.");
            int hits = 0;
            while (hits < hits_count && !validTargets.isEmpty()) {
                int randomIndex = this.rand.nextInt(validTargets.size());
                System.out.println("Selected enemy: " + randomIndex);

                Enemy target = validTargets.get(randomIndex);
                int defenseRoll = this.rand.nextInt(target.def_pts + 1);
                System.out.println("defender def roll: " + defenseRoll);

                boolean killed = Attack(target, spell_power, defenseRoll);
                if (killed) {
                    this.GainExp(target.GetExpVal());
                    validTargets.remove(randomIndex);
                }
                hits++;
            }
        }
        else {
            sendMsg(this.name + " tried to cast Blizzard but lacks mana (" + current_mana + "/" + mana_cost + ").");
            return false;
        }
        return true; 
    }
}

public class Monster extends Enemy {
    private int vision_range;

    public Monster(char tile, String name, int hp_pool, int atk_pts, int def_pts, int exp_val, int vision_range) {
        super(tile, name, hp_pool, atk_pts, def_pts, exp_val);
        this.vision_range = vision_range;
    }

    @Override
    public void processTurn(GameBoard board, Hero player) {
        Position currentPos = board.getPositionOf(this);
        Position targetPos = this.Move(currentPos, board.getPositionOf(player));
        board.attemptMove(this, targetPos);
    }

    public Position Move(Position myPos, Position playerPos) {
        double range = myPos.Range(playerPos);
        if (range < vision_range) {
            int dx = myPos.GetX() - playerPos.GetX();
            int dy = myPos.GetY() - playerPos.GetY();
            if (Math.abs(dx) > Math.abs(dy)) {
                if (dx > 0) return new Position(myPos.GetX() - 1, myPos.GetY());
                else return new Position(myPos.GetX() + 1, myPos.GetY());
            } else {
                if (dy > 0) return new Position(myPos.GetX(), myPos.GetY() - 1);
                else return new Position(myPos.GetX(), myPos.GetY() + 1);
            }
        } 
        else {
            int move = this.rand.nextInt(5);
            return switch (move) {
                case 0 -> new Position(myPos.GetX() - 1, myPos.GetY());
                case 1 -> new Position(myPos.GetX() + 1, myPos.GetY());
                case 2 -> new Position(myPos.GetX(), myPos.GetY() - 1);
                case 3 -> new Position(myPos.GetX(), myPos.GetY() + 1);
                default -> myPos;
            };
        }
    }

    public int GetVisionRange() { 
        return vision_range; 
    }
}

public class Rogue extends Hero {
    private final int MAX_ENERGY = 100;
    private int energy;
    private int cost;

    public Rogue(String name, int hp_pool, int atk_pts, int def_pts, int cost) {
        super(name, hp_pool, atk_pts, def_pts);
        energy = MAX_ENERGY;
        this.cost = cost;
        ability_range = 2;
    }

    @Override
    public void GainExp(int exp) {
        int current_lvl = this.player_lvl;
        super.GainExp(exp);
        if(current_lvl < this.player_lvl) {
             energy = MAX_ENERGY;
        }
        while(current_lvl < this.player_lvl) {
            current_lvl++;
            this.atk_pts += 3*current_lvl;
        }
    }

    @Override
    public void Tick() {
        super.Tick();
        energy = Math.min(energy + 10, MAX_ENERGY);
    }

    @Override
    public String description() {
        return super.description() + String.format(" | Energy: %d/%d", energy, MAX_ENERGY);
    }

public class Trap extends Enemy {
    private int visibility_time;
    private int invisibility_time;
    private int ticks_count;
    private boolean visible;
    private boolean triggered = false;
    private static final int TRAP_RANGE = 2;

    public Trap(char tile, String name, int hp_pool, int atk_pts, int def_pts, int exp_val, int visibility_time, int invisibility_time) {
        super(tile, name, hp_pool, atk_pts, def_pts, exp_val);
        this.visibility_time = visibility_time;
        this.invisibility_time = invisibility_time;
        this.ticks_count = 0;
        this.visible = true;
    }

    public boolean isTriggered() { return triggered; }

    @Override
    public void processTurn(GameBoard board, Hero player) {
        Position currentPos = board.getPositionOf(this);
        this.tickAndTrigger(player, currentPos, board.getPositionOf(player));
    }

    public void tickAndTrigger(Hero player, Position trapPos, Position playerPos) {
        visible = ticks_count < visibility_time;
        if (ticks_count == (visibility_time + invisibility_time)) {
            ticks_count = 0;
        } else {
            ticks_count++;
        }

        if (!triggered && trapPos.Range(playerPos) < TRAP_RANGE) {
            sendMsg("You triggered " + this.name + "!");
            this.Combat(player);
            this.triggered = true;
            this.hp_current = 0;
            player.GainExp(this.exp_val);
        }
    }

    @Override
    public void Tick() {
        visible = ticks_count < visibility_time;
        if (ticks_count == (visibility_time + invisibility_time)) {
            ticks_count = 0;
        } else {
            ticks_count++;
        }
    }

    @Override
    public String toString() {
        return (visible && !triggered) ? String.valueOf(super.tile) : ".";
    }
}

public abstract class Unit {
    protected String name;
    protected int hp_pool;
    protected int hp_current;
    protected int atk_pts;
    protected int def_pts;
    protected MessageCallback msgCallback;
    protected Random rand = new Random();

    public Unit(String name, int hp_pool, int atk_pts, int def_pts) {
        this.name = name;
        this.hp_pool = hp_pool;
        this.hp_current = hp_pool;
        this.atk_pts = atk_pts;
        this.def_pts = def_pts;
    }

    public void setMessageCallback(MessageCallback msgCallback) {
        this.msgCallback = msgCallback;
    }
    public void setRand(Random rand) {
        this.rand = rand;
    }

    protected void sendMsg(String msg) {
        if (this.msgCallback != null) {
            this.msgCallback.send(msg);
        }
    }

    //This method defines the behavior of any unit after a Game Tick.
    public void Tick() {}

    //This method defines the behavior of any unit when it engages in combat with another unit.
    //This function returns true if the target was killed in combat, and false otherwise.
    public boolean Combat(Unit target) {
        sendMsg(this.name + " engaged in combat with " + target.name + ".");
        sendMsg(this.description());
        sendMsg(target.description());

        int atk_roll = rollWithinRange(0, this.atk_pts + 1);
        int def_roll = rollWithinRange(0, target.def_pts + 1);

        sendMsg(this.name + " rolled " + atk_roll + " attack points.");
        sendMsg(target.name + " rolled " + def_roll + " defense points.");
        return Attack(target, atk_roll, def_roll); //true == killed target
    }

    public boolean Attack(Unit target, int atk_pts, int def_pts) {
        int damage = Math.max(0, atk_pts - def_pts);
        sendMsg(this.name + " dealt " + damage + " damage to " + target.name + ".");
        target.hp_current -= damage;
        if (target.hp_current <= 0) {
            target.hp_current = 0;
            sendMsg(target.name + " died.");
            return true;
        }
        return false;
    }

    public boolean isAlive() {
        return this.hp_current > 0;
    }

    protected int rollWithinRange(int min, int max) {
        return rand.nextInt(max - min + 1) + min;
    }

    public abstract String description();

    public abstract void accept(OccupantVisitor visitor);
}

            this.hp_pool += 5*current_lvl;

            this.atk_pts += 2*current_lvl;

import java.util.IdentityHashMap;

import roguelike.model.units.Enemy;
import roguelike.model.units.Hero;
import roguelike.model.units.Occupant;
import roguelike.model.units.OccupantVisitor;
import roguelike.model.units.Unit;

public class GameBoard {
    private final IdentityHashMap<Unit, Position> unitPositions = new IdentityHashMap<>();
    private final Cell[][] board;

    public GameBoard(Cell[][] board) {
        this.board = board;

        //Initialize the Unit->Position mapping
        for (int y = 0; y < board.length; y++) {
            for (int x = 0; x < board[y].length; x++) {
                Occupant occ = board[y][x].GetOccupant();
                if (occ != null && occ.GetUnit() != null) {
                    unitPositions.put(occ.GetUnit(), new Position(x, y));
                }
            }
        }
    }

    public Cell getCell(Position p) {
        return board[p.GetY()][p.GetX()];
    }

    public Occupant getOccupant(Position p) {
        return getCell(p).GetOccupant();
    }

    public void setOccupant(Position p, Occupant o) {
        getCell(p).accept(new CellVisitor() {
            @Override
            public void visit(Wall wall) {
            }
            
            @Override
            public void visit(Floor floor) {
                //remove from map if needed, update board, then update map if needed

                Unit unit = floor.GetOccupant().GetUnit();
                // Removal from map should occur only when Cell unit position and map unit position match, to avoid removing a unit that has moved to a new position
                if(p.equals(unitPositions.get(unit))) {
                    unitPositions.remove(unit);
                }
                floor.SetOccupant(o);
                // Add to map only if the new occupant has a unit (not an empty floor)
                if (o.GetUnit() != null) {
                    unitPositions.put(o.GetUnit(), p);
                }
            }
        });
    }

    public void attemptMove(Unit movingUnit, Position targetPos) {
        Position currentPos = getPositionOf(movingUnit);
        if (currentPos.equals(targetPos)) {
            return;
        }
        Cell targetCell = getCell(targetPos);
        targetCell.accept(new CellVisitor() {
            @Override
            public void visit(Wall wall) {
            }
            
            @Override
            public void visit(Floor floor) {
                handleFloorInteraction(floor, movingUnit, targetPos);
            }
        });
    }

    private void handleFloorInteraction(Floor floor, Unit movingUnit, Position targetPos) {
        if (floor.GetIsFree()) {
            Position currentPos = getPositionOf(movingUnit); 
            executeMovement(movingUnit, currentPos, targetPos);
        } else {
            Unit targetUnit = floor.GetOccupant().GetUnit();
            resolveEncounter(movingUnit, targetUnit);
        }
    }

    private void resolveEncounter(Unit attacker, Unit defender) {
        Position attackerPos = getPositionOf(attacker);
        Position defenderPos = getPositionOf(defender);

        defender.accept(new OccupantVisitor() {

            @Override
            public void visit(Hero targetHero) {
                attacker.accept(new OccupantVisitor() {
                    @Override public void visit(Hero movingHero) {}
                    @Override public void visit(Enemy movingEnemy) {
                        movingEnemy.Combat(targetHero);
                    }
                });
            }

            @Override
            public void visit(Enemy targetEnemy) {
                attacker.accept(new OccupantVisitor() {
                    @Override public void visit(Hero movingHero) {
                        boolean killed = movingHero.Combat(targetEnemy);
                        if (killed) {
                            movingHero.GainExp(targetEnemy.GetExpVal());
                            executeMovement(movingHero, attackerPos, defenderPos);
                        }
                    }
                    @Override public void visit(Enemy movingEnemy) {}
                });
            }
        });
    }

    private void executeMovement(Unit unit, Position from, Position to) {
        setOccupant(from, new Occupant());
        Occupant newOccupant = new Occupant();
        newOccupant.SetUnit(unit);
        setOccupant(to, newOccupant);
    }

    public Position getPositionOf(Unit unit) {
        return unitPositions.get(unit);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (int y = 0; y < board.length; y++) {
            for (int x = 0; x < board[y].length; x++) {
                sb.append(board[y][x].toString());
            }
            sb.append("\n");
        }
        return sb.toString();
    }
}

import java.util.Objects;

public final class Position {
    private final int x;
    private final int y;

    public Position(int x, int y) {
        this.x = x;
        this.y = y;
    }

    public int GetX() { return x; }
    public int GetY() { return y; }

    public double Range(Position other) {
        int dx = this.x - other.x;
        int dy = this.y - other.y;
        return Math.sqrt(dx*dx + dy*dy);
    }

    @Override 
    public boolean equals(Object other) {
        if(this == other) return true;
        if(!(other instanceof Position)) return false;
        Position otherPos = (Position) other;
        return this.x == otherPos.x && this.y == otherPos.y;
    }

    @Override 
    public int hashCode() {
        return Objects.hash(x, y);
    }

    @Override 
    public String toString() {
        return String.format("(%d, %d)", x, y);
    }
}

package roguelike.world;

import java.util.Random;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import roguelike.model.units.*;

import static org.junit.jupiter.api.Assertions.*;

public class GameBoardTest {
    private GameBoard board;
    private Warrior player;

    @BeforeEach
    public void setUp() {
        Cell[][] cells = new Cell[3][3];
        for (int y = 0; y < 3; y++) {
            for (int x = 0; x < 3; x++) {
                if (x == 1 && y == 0) {
                    cells[y][x] = new Wall(new Position(x, y));
                } else {
                    cells[y][x] = new Floor(new Position(x, y));
                }
            }
        }
        board = new GameBoard(cells);
        player = new Warrior("Test Player", 100, 10, 5, 3);
        player.setMessageCallback(msg -> {});
    }

    @Test
    public void testMoveToEmptyFloor() {
        Occupant occ = new Occupant();
        occ.SetUnit(player);
        board.setOccupant(new Position(0, 0), occ); 

        Position targetPos = new Position(0, 1); 
        board.attemptMove(player, targetPos);

        assertNull(board.getOccupant(new Position(0, 0)).GetUnit(), "Old position empty");
        assertEquals(player, board.getOccupant(targetPos).GetUnit(), "Player at new position");
    }

    @Test
    public void testMoveToWallIsBlocked() {
        Occupant occ = new Occupant();
        occ.SetUnit(player);
        board.setOccupant(new Position(0, 0), occ);

        Position targetPos = new Position(1, 0); 
        
        final boolean[] hitWall = {false};
        board.getCell(targetPos).accept(new CellVisitor() {
            @Override
            public void visit(Wall wall) { hitWall[0] = true; }
            @Override
            public void visit(Floor floor) { board.attemptMove(player, targetPos); }
        });

        assertTrue(hitWall[0], "Visitor identified wall");
        assertEquals(player, board.getOccupant(new Position(0, 0)).GetUnit(), "Player remained in place");
    }

    private Occupant occupantOf(Unit unit) {
        Occupant occ = new Occupant();
        occ.SetUnit(unit);
        return occ;
    }

    @Test
    public void placedUnitIsFound() {
        board.setOccupant(new Position(0, 0), occupantOf(player));
        assertEquals(new Position(0, 0), board.getPositionOf(player));
    }

    @Test
    public void neverPlacedUnitReturnsNull() {
        Warrior stranger = new Warrior("Stranger", 100, 10, 5, 3);
        assertNull(board.getPositionOf(stranger));
    }

    @Test
    public void movedUnitHasNewPosition() {
        board.setOccupant(new Position(0, 0), occupantOf(player));
        board.attemptMove(player, new Position(0, 1));
        assertEquals(new Position(0, 1), board.getPositionOf(player));
    }

    @Test
    public void removedUnitReturnsNull() {
        Position pos = new Position(0, 0);
        board.setOccupant(pos, occupantOf(player));
        board.setOccupant(pos, new Occupant());
        assertNull(board.getPositionOf(player));
    }

    @Test
    public void seededRandProducesDeterministicCombat() {
        Warrior attacker1 = new Warrior("Attacker", 100, 10, 5, 3);
        Warrior attacker2 = new Warrior("Attacker", 100, 10, 5, 3);

        Monster defender1 = new Monster('m', "Defender", 5, 5, 10, 3, 1);
        Monster defender2 = new Monster('m', "Defender", 5, 5, 10, 3, 1);


        Random rand = new Random(25); // fixed seed for deterministic test
        attacker1.setRand(rand);
        attacker2.setRand(rand);

        boolean result1 = attacker1.Combat(defender1);
        boolean result2 = attacker2.Combat(defender2);
        assertEquals(result1, result2, "Combat results should be the same with the same seed");
        assertEquals(defender1.description(), defender2.description(), "Defender states should be the same with the same seed");
    }

    @Test 
    public void guaranteedKillAndGuaranteedMiss() {
        Warrior attacker1 = new Warrior("Attacker", 100, 1_000_000, 0, 3);
        Warrior attacker2 = new Warrior("Attacker", 100, 10, 0, 3);
        
        Monster defender1 = new Monster('m', "Defender", 1, 1, 0, 3, 1);
        Monster defender2 = new Monster('m', "Defender", 100, 1, 1_000_000, 3, 1);

        Random rand = new Random(42); // fixed seed for deterministic test
        attacker1.setRand(rand);

        boolean result1 = attacker1.Combat(defender1);
        assertTrue(result1, "Defender should die after combat - 1 in a million chance to miss");

        String defender2_desc_before = defender2.description();
        attacker2.setRand(rand);

        boolean result2 = attacker2.Combat(defender2);
        assertFalse(result2, "Defender should survive after combat - 1 in a million chance to hit");
        assertEquals(defender2_desc_before, defender2.description(), "Defender state should be unchanged after combat");
    }

    @Test
    public void killingAnEnemyAndMovingIntoItsCellUpdatesBothUnits() {
        // Huge attack against 1 HP and 0 DEF: the kill is effectively guaranteed
        // The random seed is fixed to ensure deterministic behavior in the test
        Warrior killer = new Warrior("Killer", 100, 1_000_000, 0, 3);
        killer.setMessageCallback(msg -> {});
        Monster victim = new Monster('m', "Victim", 1, 0, 0, 0, 1);
        victim.setMessageCallback(msg -> {});

        Random rand = new Random(42); // fixed seed for deterministic test
        killer.setRand(rand);
        victim.setRand(rand);

        board.setOccupant(new Position(0, 0), occupantOf(killer));
        board.setOccupant(new Position(0, 1), occupantOf(victim));

        board.attemptMove(killer, new Position(0, 1));

        assertNull(board.getPositionOf(victim), "Dead enemy should leave the map");
        assertEquals(new Position(0, 1), board.getPositionOf(killer));
    }

    @Test
    public void staleClearDoesNotRemoveMovedUnit() {
        Position from = new Position(0, 0);
        Position to = new Position(0, 1);

        board.setOccupant(from, occupantOf(player));
        // wrong order on purpose: fill the new cell first, clear the old cell second
        board.setOccupant(to, occupantOf(player));
        board.setOccupant(from, new Occupant());

        assertEquals(to, board.getPositionOf(player), "Stale clear must not delete the new entry");
    }
}
